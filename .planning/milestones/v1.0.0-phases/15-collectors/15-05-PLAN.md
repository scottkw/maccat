  Imports: from __future__ import annotations; stdlib: import json, re, shutil, subprocess from pathlib import Path; maccat: from maccat.catalog.format import emit_item; from maccat.collectors.base import Collector, CollectorResult, Section.

  class CodexCollector(Collector):

    def _collect_via_cli(self) -> list[str]:
      subprocess.run(["codex", "mcp", "list", "--json"], capture_output=True, text=True, shell=False).
      Return [] if returncode != 0 or stdout.strip() is empty.
      Try json.loads(stdout); if not isinstance(entries, list) or not entries: return [].
      For each entry: name = entry.get("name", ""); transport = entry.get("type", "stdio").
      # CAT-05: ONLY .name and .type — never .command, .env, .args, .url, .headers
      Clamp transport via _TRANSPORT_WHITELIST. line = emit_item(name, "", transport); collect.
      Return items. Catch json.JSONDecodeError → return [].

    def _collect_via_toml(self) -> list[str]:
      Try _TOML_PATH.read_text(encoding="utf-8"). Except OSError: return [].
      # CAT-05 + Pitfall G: read ONLY section header lines via regex — NEVER tomllib
      For each line in text.splitlines():
        m = re.match(r'^\[mcp_servers\.(.*)\]$', line.strip())
        if m: name = m.group(1).strip('"'); transport = "stdio"  # default, value lines never read
        line = emit_item(name, "", transport); if line: items.append(line)
      Return items.

    def collect(self) -> CollectorResult:
      items = [].
      If shutil.which("codex"): items = self._collect_via_cli().
      If not items and _TOML_PATH.is_file(): items = self._collect_via_toml().
      Return CollectorResult(sections=[Section(title=_TITLE, items=items)]).

  __all__ = ["CodexCollector"]

  Imports: from __future__ import annotations; stdlib: import json, sys from pathlib import Path; maccat: from maccat.catalog.format import emit_item; from maccat.collectors.base import Collector, CollectorResult, Section; from maccat.collectors.claude import _read_yaml_name.

  Module-level constants:
    _CONFIG_PATH = Path.home() / ".config/opencode/opencode.json"
    _AGENTS_DIR = Path.home() / ".config/opencode/agents"
    _TRANSPORT_WHITELIST = frozenset({"stdio", "http", "sse"})

  class OpenCodeCollector(Collector):

    def _load_config(self) -> dict | None:
      If not _CONFIG_PATH.is_file(): return None.
      Try json.loads(_CONFIG_PATH.read_text(encoding="utf-8")). Except (json.JSONDecodeError, OSError): return None.

    def _collect_plugins(self) -> Section:
      title = "OpenCode Plugins".
      data = self._load_config(); if data is None: return Section(title=title, items=[]).
      items = [].
      For entry in (data.get("plugin") or []):
        name = entry.split("@", 1)[0]
        # Path/URL guard: no '@' AND contains '/' → warn to stderr, skip
        if name == entry and "/" in entry:
          print(f"  WARNING: skipping OpenCode plugin path/URL: {entry}", file=sys.stderr); continue
        if not name: continue
        line = emit_item(name, "", ""); if line: items.append(line)
      Return Section(title=title, items=items).

    def _collect_mcp(self) -> Section:
      title = "OpenCode MCP Servers".
      data = self._load_config(); if data is None or not data.get("mcp"): return Section(title=title, items=[]).
      items = [].
      For name, cfg in data["mcp"].items():
        # CAT-05: ONLY .type — NEVER .command, .env, .args, .url, .headers
        transport = cfg.get("type", "stdio")
        if transport not in _TRANSPORT_WHITELIST: transport = "stdio"
        line = emit_item(name, "", transport); if line: items.append(line)
      Return Section(title=title, items=items).

    def _collect_agents(self) -> Section:
      title = "OpenCode Agents".
      If not _AGENTS_DIR.is_dir(): return Section(title=title, items=[]).
      items = [].
      For agent_md in sorted(_AGENTS_DIR.glob("*.md")):
        name = _read_yaml_name(agent_md)
        if not name: name = agent_md.stem
        line = emit_item(name, "", ""); if line: items.append(line)
      Return Section(title=title, items=items).

    def collect(self) -> CollectorResult:
      Return CollectorResult(sections=[self._collect_plugins(), self._collect_mcp(), self._collect_agents()]).

  __all__ = ["OpenCodeCollector"]
