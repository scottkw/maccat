  Module docstring: "HomebrewCollector — raw-write byte-parity with update-list.sh:2233 (generate_catalog Homebrew section)."
  Imports: from __future__ import annotations; import shutil, subprocess, sys from stdlib; from maccat.collectors.base import Collector, CollectorResult, Section.

  TITLE = "Homebrew Packages" (class constant).

  available(): shutil.which("brew") is not None.

  Private _run(cmd: list[str]) -> list[str]: subprocess.run(cmd, capture_output=True, text=True, shell=False). Returns [] on returncode != 0. Returns result.stdout.rstrip("\n").split("\n") if result.stdout.strip() else []. Never raises.

  collect(): If not available(), print WARNING to sys.stderr ("  WARNING: brew not found."), return CollectorResult(sections=[Section(title=TITLE, items=["Homebrew is not installed."], raw=True)]). Otherwise: formulae = _run(["brew", "list", "--formula"]), casks = _run(["brew", "list", "--cask"]), lines = formulae + casks. Return CollectorResult(sections=[Section(title=TITLE, items=lines, raw=True)]). NOTE: raw=True on every Section — orchestrator writes with write_lines(), never flush_section().

  Create src/maccat/collectors/mas.py:

  Module docstring: "MasCollector — raw-write byte-parity with update-list.sh:2249 (generate_catalog App Store section)."
  Imports: from __future__ import annotations; import shutil, subprocess, sys from stdlib; from maccat.collectors.base import Collector, CollectorResult, Section.

  TITLE = "App Store Applications".

  available(): shutil.which("mas") is not None.

  Private _parse_mas_output(stdout: str) -> list[str]: Split each line on whitespace. If parts has length >= 3, output f"{parts[1]} {parts[2]}". This mirrors `awk '{print $2, $3}'` — column 1 is the numeric App Store ID (skipped), columns 2 and 3 are AppName and (version).

  collect():
    If not available(): print WARNING to sys.stderr, return CollectorResult with Section(title=TITLE, items=["mas (Mac App Store CLI) is not installed.", "Install it with Homebrew: brew install mas"], raw=True).
    result = subprocess.run(["mas", "list"], capture_output=True, text=True).
    If result.returncode != 0: return CollectorResult with Section(title=TITLE, items=["Could not retrieve App Store list."], raw=True).
    lines = _parse_mas_output(result.stdout).
    Return CollectorResult(sections=[Section(title=TITLE, items=lines, raw=True)]).
  </action>
