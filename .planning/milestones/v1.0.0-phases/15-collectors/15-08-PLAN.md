    _BASE = Path.home() / "Library/Application Support/Google/Chrome"
    _TITLE = "Google Chrome Extensions"

  class ChromeCollector(Collector):

    def _collect_profile(self, extensions_dir: Path) -> list[str]:
      items = [].
      For ext_dir in extensions_dir.iterdir():
        if not ext_dir.is_dir(): continue
        ext_id = ext_dir.name
        if ext_id == "Temp" or ext_id.startswith("_") or ext_id in COMPONENT_DENYLIST: continue
        candidates = [d.name for d in ext_dir.iterdir() if d.is_dir()]
        ver_dir = version_sort_tail(candidates)  # Phase 13 helper — sort -V | tail -1
        if not ver_dir: continue
        manifest = ext_dir / ver_dir / "manifest.json"
        if not manifest.is_file(): continue
        name = chrome_ext_name(manifest)  # Phase 13 helper — resolves __MSG_key__
        version = json_get(manifest, "version")  # Phase 13 helper
        line = emit_item(name, version, ext_id); if line: items.append(line)
      Return items.

    def collect(self) -> CollectorResult:
      If not _BASE.is_dir():
        print("  NOTE: Google Chrome not installed.", file=sys.stderr)
        Return CollectorResult(sections=[Section(title=_TITLE, items=[])]).
      all_items = [].
      # Profile enumeration: Default first, then sorted "Profile */" dirs (zsh :2089 order)
      profile_dirs = [_BASE / "Default"]
      profile_dirs += sorted(_BASE.glob("Profile */"))
      For profile in profile_dirs:
        ext_root = profile / "Extensions"
        if not ext_root.is_dir(): continue
        all_items.extend(self._collect_profile(ext_root))
      Return CollectorResult(sections=[Section(title=_TITLE, items=all_items)]).
      # NOTE: items are raw unsorted emit_item lines; Phase 16 calls flush_section(items) before writing.
      # raw=False (default) — flush_section handles cross-profile dedup via LC_ALL=C sort -f -u.

  __all__ = ["ChromeCollector", "COMPONENT_DENYLIST"]
