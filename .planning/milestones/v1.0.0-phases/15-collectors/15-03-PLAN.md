  Module docstring: "SetappCollector — raw-write byte-parity with update-list.sh:2267 (generate_catalog Setapp section)."
  Imports: from __future__ import annotations; from pathlib import Path; from maccat.collectors.base import Collector, CollectorResult, Section.

  TITLE = "Setapp Applications"
  BASE = Path("/Applications/Setapp")

  available(): BASE.is_dir().

  collect():
    If not available(): return CollectorResult(sections=[Section(title=TITLE, items=["Setapp is not installed or detected."], raw=True)]).
    PITFALL C: include the root dir itself (find includes start path).
    entries: list[str] = [BASE.name]  # "Setapp" — the start path that find produces
    entries += [p.name for p in BASE.iterdir() if p.is_dir()]
    entries.sort()
    Return CollectorResult(sections=[Section(title=TITLE, items=entries, raw=True)]).

  Create src/maccat/collectors/webapps.py:

  Module docstring: "WebAppsCollector — raw-write byte-parity with update-list.sh:2281 (generate_catalog Web-installed section)."
  Imports: from __future__ import annotations; import fnmatch; from pathlib import Path; from maccat.collectors.base import Collector, CollectorResult, Section.

  TITLE = "Web-installed Applications"
  BASE = Path("/Applications")

  available(): always True — no guard (WebAppsCollector has no availability check; /Applications always exists on macOS). Do NOT override available().

  collect():
    PITFALL C: include the root dir itself ("Applications").
    entries: list[str] = [BASE.name]  # "Applications" — the start path that find produces
    for p in BASE.iterdir():
        if not p.is_dir(): continue
        if fnmatch.fnmatch(p.name, "Setapp*"): continue
        if fnmatch.fnmatch(p.name, "*App Store*"): continue
        entries.append(p.name)
    entries.sort()
    Return CollectorResult(sections=[Section(title=TITLE, items=entries, raw=True)]).

  Both collectors: raw=True on all Sections. No emit_item. No flush_section. No subprocess calls.
