  Module docstring: "GeminiCollector — 2-section collector at byte-parity with update-list.sh:1970/2016 (collect_gemini_extensions/collect_gemini_mcp). CAT-05 on MCP; Pitfall B empty-file guard on mcp_config.json."

  Imports: from __future__ import annotations; stdlib: import json from pathlib import Path; maccat: from maccat.catalog.format import emit_item; from maccat.collectors.base import Collector, CollectorResult, Section; from maccat.helpers.json_io import json_get.

  Module-level constants:
    _EXT_DIR = Path.home() / ".gemini/extensions"
    _MCP_PATH = Path.home() / ".gemini/config/mcp_config.json"
    _EXT_TITLE = "Gemini CLI Extensions"
    _MCP_TITLE = "Gemini CLI MCP Servers"
    _TRANSPORT_WHITELIST = frozenset({"stdio", "http", "sse"})

  class GeminiCollector(Collector):

    def _collect_extensions(self) -> Section:
      If not _EXT_DIR.is_dir(): return Section(title=_EXT_TITLE, items=[]).
      items = [].
      For ext_dir in sorted(_EXT_DIR.iterdir()) where ext_dir.is_dir():
        manifest = ext_dir / "gemini-extension.json"
        if not manifest.is_file(): continue
        name = json_get(manifest, "name") or ext_dir.name  # fallback to basename if name absent/empty
        version = json_get(manifest, "version")
        line = emit_item(name, version, ""); if line: items.append(line)
      Return Section(title=_EXT_TITLE, items=items).

    def _collect_mcp(self) -> Section:
      # Pitfall B: use [[ -s ]] equivalent — is_file() AND stat().st_size > 0
      # A plain is_file() check returns True for a 0-byte file; json.loads("") raises JSONDecodeError
      if not _MCP_PATH.is_file() or _MCP_PATH.stat().st_size == 0:
        return Section(title=_MCP_TITLE, items=[]).
      Try json.loads(_MCP_PATH.read_text(encoding="utf-8")). Except (json.JSONDecodeError, OSError): return Section(title=_MCP_TITLE, items=[]).
      items = [].
      For name, cfg in (data.get("mcpServers") or {}).items():
        # CAT-05: ONLY .type — NEVER .command, .env, .args, .url, .headers
        transport = cfg.get("type", "stdio")
        if transport not in _TRANSPORT_WHITELIST: transport = "stdio"
        line = emit_item(name, "", transport); if line: items.append(line)
      Return Section(title=_MCP_TITLE, items=items).

    def collect(self) -> CollectorResult:
      Return CollectorResult(sections=[self._collect_extensions(), self._collect_mcp()]).

  __all__ = ["GeminiCollector"]
