  Imports: from __future__ import annotations; stdlib: import json, re from pathlib import Path; maccat: from maccat.catalog.format import emit_item; from maccat.collectors.base import Collector, CollectorResult, Section.

  Module-level helper (not a method — used by both ClaudeCollector and shared via other collectors in later plans):
    def _read_yaml_name(path: Path) -> str:
      Reads first line starting with "name:" from path. Strips "name:" prefix, strips leading/trailing whitespace, strips surrounding double-quotes. Returns "" on OSError. Mirrors: grep '^name:' FILE | head -1 | sed 's/^name:[[:space:]]*//' | tr -d '"'

  class ClaudeCollector(Collector):

    _PLUGINS_TITLE = "Claude Code Plugins"
    _MCP_TITLE = "Claude Code MCP Servers"
    _SKILLS_TITLE = "Claude Code Skills & Agents"

    def _collect_plugins(self) -> Section:
      If _PLUGINS_PATH not is_file(): return Section(title=_PLUGINS_TITLE, items=[]).
      Try json.loads(_PLUGINS_PATH.read_text(encoding="utf-8")). Except (json.JSONDecodeError, OSError): return Section(title=_PLUGINS_TITLE, items=[]).
      For key, versions in (data.get("plugins") or {}).items():
        name = key.split("@", 1)[0]
        version = versions[0].get("version", "") if versions else ""
        line = emit_item(name, version, key)
        if line: items.append(line)
      Return Section(title=_PLUGINS_TITLE, items=items). raw=False (default) — these go through flush_section.

    def _collect_mcp(self) -> Section:
      If _CLAUDE_JSON not is_file(): return Section(title=_MCP_TITLE, items=[]).
      Try json.loads(_CLAUDE_JSON.read_text(encoding="utf-8")). Except (json.JSONDecodeError, OSError): return Section(title=_MCP_TITLE, items=[]).
      For name, cfg in (data.get("mcpServers") or {}).items():
        # CAT-05: ONLY .type — NEVER .command, .env, .args, .url, .headers
        transport = cfg.get("type", "stdio")
        if transport not in _TRANSPORT_WHITELIST: transport = "stdio"
        line = emit_item(name, "", transport)
        if line: items.append(line)
      Return Section(title=_MCP_TITLE, items=items).

    def _collect_skills_agents(self) -> Section:
      items = [].
      If _SKILLS_DIR.is_dir():
        For skill_dir in sorted(_SKILLS_DIR.iterdir()) where skill_dir.is_dir():
          skill_md = skill_dir / "SKILL.md"
          name = _read_yaml_name(skill_md) if skill_md.is_file() else ""
          if not name: name = skill_dir.name
          line = emit_item(name, "", ""); if line: items.append(line)
      If _AGENTS_DIR.is_dir():
        For agent_md in sorted(_AGENTS_DIR.glob("*.md")):
          name = _read_yaml_name(agent_md)
          if not name: name = agent_md.stem
          line = emit_item(name, "", ""); if line: items.append(line)
      Return Section(title=_SKILLS_TITLE, items=items).

    def collect(self) -> CollectorResult:
      Return CollectorResult(sections=[self._collect_plugins(), self._collect_mcp(), self._collect_skills_agents()]).

  __all__ = ["ClaudeCollector"]
