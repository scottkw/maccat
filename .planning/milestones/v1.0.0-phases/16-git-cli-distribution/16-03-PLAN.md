    Write tests/test_pyz.py. This file is a smoke-test suite that verifies the built .pyz artifact.

    Imports: from __future__ import annotations; import os, subprocess, sys; from pathlib import Path; import pytest.

    test_pyz_version_from_unrelated_cwd(tmp_path: Path):
    - _require_pyz()
    - result = subprocess.run([sys.executable, str(PYZ), "--version"], cwd=str(tmp_path), capture_output=True, text=True)
    - assert result.returncode == 0
    - assert "maccat" in result.stdout
    Acceptance: PKG-03 — .pyz runs correctly from an unrelated cwd (success criterion 1)

    test_pyz_help_from_unrelated_cwd(tmp_path: Path):
    - _require_pyz()
    - result = subprocess.run([sys.executable, str(PYZ), "--help"], cwd=str(tmp_path), capture_output=True, text=True)
    - assert result.returncode == 0
    Acceptance: PKG-05 — --help works from any directory

    test_pyz_no_so_dylib():
    - _require_pyz()
    - import zipfile; zf = zipfile.ZipFile(str(PYZ))
    - names = zf.namelist(); zf.close()
    - bad = [n for n in names if n.endswith(".so") or n.endswith(".dylib")]
    - assert bad == [], f"Unexpected native libs in .pyz: {bad}"
    Acceptance: PKG-03 — no .so/.dylib files (pure Python only, no C extensions)

    test_pyz_no_file_relative_catalog(tmp_path: Path):
    - _require_pyz()
    - Build a stripped env: keep only PATH and TERM, explicitly exclude MACCAT_CATALOG_DIR and XDG_CONFIG_HOME to guarantee no config.toml is found
        env = {k: v for k, v in os.environ.items() if k in ("PATH", "TERM", "HOME")}
    - Also pass HOME pointing to tmp_path so no real ~/.config/maccat/config.toml is found:
        env["HOME"] = str(tmp_path)
    - result = subprocess.run([sys.executable, str(PYZ)], cwd=str(tmp_path), env=env, capture_output=True, text=True)
    - assert result.returncode != 0  # must fail (no catalog configured)
    - output = (result.stdout + result.stderr).lower()
    - assert "catalog" in output   # must print actionable message, not silent __file__ fallback
    Acceptance: PKG-03 — catalog_repo never resolved from __file__ (success criterion 1)

    test_pyz_maccat_package_importable_from_pyz():
    - _require_pyz()
    - result = subprocess.run([sys.executable, "-c", f"import sys; sys.path.insert(0, '{PYZ}'); import maccat; print(maccat.__version__)"], capture_output=True, text=True)
    - assert "1.0.0" in result.stdout
    Acceptance: the maccat package is correctly bundled at the maccat/ path (not at archive root)
